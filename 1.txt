"""Forward-only UMM CE scores used for ReCA trajectory selection."""

import torch
import torch.nn.functional as F

from data.data_utils import patchify
from .qwen2_navit import NaiveCache


MATCH_TEMPLATE = (
    "Compare the image with this target description: {prompt}\n"
    "Answer exactly Match if all required content agrees, otherwise answer "
    "Mismatch."
)


def _token_id(new_token_ids, name):
    value = new_token_ids[name]
    return int(value.item()) if torch.is_tensor(value) else int(value)


@torch.no_grad()
def score_user_match_mismatch(
    model,
    latent,
    image_shape,
    vae_model,
    image_transform,
    tokenizer,
    new_token_ids,
    user_prompt,
    vit_max_side=224,
):
    """Return NLL(Match)-NLL(Mismatch) for one look-ahead latent.

    Match and Mismatch share one image/instruction forward.  Lower is better.
    """
    height, width = image_shape
    latent_h = height // model.latent_downsample
    latent_w = width // model.latent_downsample
    latent = latent.reshape(
        1,
        latent_h,
        latent_w,
        model.latent_patch_size,
        model.latent_patch_size,
        model.latent_channel,
    )
    latent = torch.einsum("nhwpqc->nchpwq", latent)
    latent = latent.reshape(
        1,
        model.latent_channel,
        latent_h * model.latent_patch_size,
        latent_w * model.latent_patch_size,
    )
    vae_device = next(vae_model.parameters()).device
    image = vae_model.decode(latent.to(device=vae_device, dtype=torch.bfloat16))
    image = (image * 0.5 + 0.5).clamp(0, 1)

    image_tensor = image[0]
    image_h, image_w = image_tensor.shape[-2:]
    if max(image_h, image_w) > vit_max_side:
        scale = vit_max_side / max(image_h, image_w)
        resized_h = max(
            model.vit_patch_size,
            round(image_h * scale / model.vit_patch_size) * model.vit_patch_size,
        )
        resized_w = max(
            model.vit_patch_size,
            round(image_w * scale / model.vit_patch_size) * model.vit_patch_size,
        )
        image_tensor = F.interpolate(
            image_tensor.unsqueeze(0),
            size=(resized_h, resized_w),
            mode="bicubic",
            align_corners=False,
            antialias=True,
        )[0]

    mean = torch.as_tensor(
        image_transform.normalize_transform.mean,
        device=image_tensor.device,
        dtype=image_tensor.dtype,
    )[:, None, None]
    std = torch.as_tensor(
        image_transform.normalize_transform.std,
        device=image_tensor.device,
        dtype=image_tensor.dtype,
    )[:, None, None]
    image_tensor = (image_tensor - mean) / std

    device = next(model.vit_model.parameters()).device
    vit_position_ids = model.get_flattened_position_ids(
        image_tensor.shape[-2],
        image_tensor.shape[-1],
        model.vit_patch_size,
        max_num_patches_per_side=model.vit_max_num_patch_per_side,
    ).to(device)
    packed_vit_tokens = patchify(image_tensor, model.vit_patch_size).to(device)
    num_vit_tokens = packed_vit_tokens.shape[0]
    vit_token_seqlens = torch.tensor(
        [num_vit_tokens], dtype=torch.int, device=device
    )
    cu_seqlens = F.pad(
        torch.cumsum(vit_token_seqlens, dim=0), (1, 0)
    ).to(torch.int32)
    packed_vit_embed = model.vit_model(
        packed_pixel_values=packed_vit_tokens,
        packed_flattened_position_ids=vit_position_ids,
        cu_seqlens=cu_seqlens,
        max_seqlen=num_vit_tokens,
    )
    packed_vit_embed = model.connector(packed_vit_embed)
    packed_vit_embed = packed_vit_embed + model.vit_pos_embed(vit_position_ids)

    boundary_ids = torch.tensor(
        [
            _token_id(new_token_ids, "start_of_image"),
            _token_id(new_token_ids, "end_of_image"),
        ],
        dtype=torch.long,
        device=device,
    )
    boundary_embed = model.language_model.model.embed_tokens(boundary_ids)
    packed_sequence = boundary_embed.new_zeros(
        (num_vit_tokens + 2, model.hidden_size)
    )
    packed_sequence[0] = boundary_embed[0]
    packed_sequence[-1] = boundary_embed[1]
    packed_sequence[1:-1] = packed_vit_embed.to(packed_sequence.dtype)

    ce_cache = NaiveCache(model.config.llm_config.num_hidden_layers)
    image_lens = torch.tensor(
        [num_vit_tokens + 2], dtype=torch.int, device=device
    )
    image_indexes = torch.arange(
        num_vit_tokens + 2, dtype=torch.long, device=device
    )
    image_positions = torch.zeros(
        num_vit_tokens + 2, dtype=torch.long, device=device
    )
    empty_kv_indexes = torch.empty(0, dtype=torch.long, device=device)
    empty_kv_lens = torch.zeros(1, dtype=torch.int, device=device)
    extra_inputs = {"mode": "und"} if model.use_moe else {}
    image_output = model.language_model.forward_inference(
        packed_query_sequence=packed_sequence,
        query_lens=image_lens,
        packed_query_position_ids=image_positions,
        packed_query_indexes=image_indexes,
        past_key_values=ce_cache,
        key_values_lens=empty_kv_lens,
        packed_key_value_indexes=empty_kv_indexes,
        update_past_key_values=True,
        is_causal=False,
        **extra_inputs,
    )

    instruction = MATCH_TEMPLATE.format(prompt=user_prompt)
    generation_input, current_lens, current_rope = model.prepare_prompts(
        curr_kvlens=[num_vit_tokens + 2],
        curr_rope=[1],
        prompts=[instruction],
        tokenizer=tokenizer,
        new_token_ids=new_token_ids,
    )
    for key, value in generation_input.items():
        if torch.is_tensor(value):
            generation_input[key] = value.to(device)
    instruction_embed = model.language_model.model.embed_tokens(
        generation_input["packed_text_ids"]
    )
    instruction_output = model.language_model.forward_inference(
        packed_query_sequence=instruction_embed,
        query_lens=generation_input["text_token_lens"],
        packed_query_position_ids=generation_input["packed_text_position_ids"],
        packed_query_indexes=generation_input["packed_text_indexes"],
        past_key_values=image_output.past_key_values,
        packed_key_value_indexes=generation_input["packed_key_value_indexes"],
        key_values_lens=generation_input["key_values_lens"],
        update_past_key_values=True,
        is_causal=True,
        **extra_inputs,
    )

    match_ids = tokenizer.encode("Match")
    mismatch_ids = tokenizer.encode("Mismatch")
    if not match_ids or not mismatch_ids:
        raise RuntimeError("Match/Mismatch verbalizers produced no tokens")
    input_ids = torch.tensor(
        [_token_id(new_token_ids, "bos_token_id")],
        dtype=torch.long,
        device=device,
    )
    current_kvlen = int(current_lens[0])
    current_position = int(current_rope[0])
    output = model.language_model.forward_inference(
        packed_query_sequence=model.language_model.model.embed_tokens(input_ids),
        query_lens=torch.ones(1, dtype=torch.int, device=device),
        packed_query_position_ids=torch.tensor(
            [current_position], dtype=torch.long, device=device
        ),
        packed_query_indexes=torch.tensor(
            [current_kvlen], dtype=torch.long, device=device
        ),
        past_key_values=instruction_output.past_key_values,
        key_values_lens=torch.tensor(
            [current_kvlen], dtype=torch.int, device=device
        ),
        packed_key_value_indexes=torch.arange(
            current_kvlen, dtype=torch.long, device=device
        ),
        update_past_key_values=False,
        is_causal=True,
        **extra_inputs,
    )
    logits = model.language_model.lm_head(output.packed_query_sequence).float()
    match_label = torch.tensor([match_ids[0]], dtype=torch.long, device=device)
    mismatch_label = torch.tensor(
        [mismatch_ids[0]], dtype=torch.long, device=device
    )
    match_nll = F.cross_entropy(logits, match_label)
    mismatch_nll = F.cross_entropy(logits, mismatch_label)
    return {
        "energy": float((match_nll - mismatch_nll).item()),
        "match_nll": float(match_nll.item()),
        "mismatch_nll": float(mismatch_nll.item()),
        "vit_tokens": int(num_vit_tokens),
    }
