"""视频 -> VAE 帧预处理 -> DistributedVAE embedding -> outdir + JSONL。

目前按 vae.py 的接口使用 DistributedVAE，未来可替换 build_encoder。
所有分布式 rank 协作编码同一个视频，仅 rank 0 写文件。
输出 .pt 是 encode(x)[0] 的结果（不额外采样 latent），排列为 [T,C,H,W]。
"""
import argparse
import datetime
import json
import os
from pathlib import Path, PurePosixPath
import pickle
import subprocess
import sys
import tempfile
from contextlib import contextmanager


VAE_TARGET_SIZES = {
    "480p": {1.0: [512, 512], 1.5: [720, 480], 0.6667: [480, 720]},
    "720p": {1.7778: [1072, 720]},
    "400p": {0.5625: [592, 400]},
    "640p": {1.0: [640, 640], 1.5: [960, 640], 0.6667: [640, 960]},
    "560p": {1.0: [576, 576], 1.5: [840, 560], 0.6667: [560, 840]},
    "320p": {1.0: [320, 320], 1.5: [480, 320], 0.6667: [320, 480]},
    "240p": {1.0: [256, 256], 1.5: [360, 240], 0.6667: [240, 360]},
}


def load_video(video_path, max_frames_num=121, fps=24, image_size="480p", ffmpeg_dir=""):
    """对齐 vae.py 的 ffmpeg-subprocess 分支，返回 RGB uint8 [T,H,W,C]。

    时间范围、FPS、缩放裁剪和帧数截断与 VAE 一致；调用方归一化到 [-1,1]。
    """
    import numpy as np

    if fps <= 0 or max_frames_num <= 0:
        raise ValueError("fps and max_frames_num must be positive")
    ffprobe = os.path.join(ffmpeg_dir, "ffprobe")
    ffmpeg = os.path.join(ffmpeg_dir, "ffmpeg")

    def probe(attribute):
        return subprocess.check_output([
            ffprobe, "-v", "error", "-select_streams", "v:0",
            "-show_entries", f"stream={attribute}",
            "-of", "default=noprint_wrappers=1:nokey=1", video_path,
            "-loglevel", "quiet",
        ], stderr=subprocess.STDOUT).decode("utf-8").strip()

    duration = float(probe("duration"))
    width, height = int(probe("width")), int(probe("height"))
    target_sizes = VAE_TARGET_SIZES[image_size]
    closest_ratio = min(target_sizes, key=lambda ratio: abs(ratio - height / width))
    crop_h, crop_w = target_sizes[closest_ratio]
    ratio = max(crop_h / height, crop_w / width)
    resize_h, resize_w = int(np.ceil(height * ratio)), int(np.ceil(width * ratio))
    end = min(duration, max_frames_num // fps + 1)
    result = subprocess.run([
        ffmpeg, "-ss", "0", "-to", str(end), "-i", video_path,
        "-vf", f"fps={fps},scale={resize_w}:{resize_h}:flags=bilinear,crop=w={crop_w}:h={crop_h}",
        "-vsync", "0", "-vcodec", "rawvideo", "-pix_fmt", "rgb24",
        "-f", "image2pipe", "-",
    ], stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    if result.returncode:
        raise ValueError(f"FFmpeg failed for {video_path}: {result.stderr.decode(errors='replace')}")
    bytes_per_frame = crop_h * crop_w * 3
    if len(result.stdout) % bytes_per_frame:
        raise ValueError(f"Incomplete RGB frame from {video_path}")
    frame_count = len(result.stdout) // bytes_per_frame
    frame_count = min(frame_count - (frame_count - 1) % 8, max_frames_num)
    if frame_count < 73:
        raise ValueError(f"VAE requires at least 73 frames, got {frame_count}: {video_path}")
    return np.frombuffer(result.stdout, dtype=np.uint8).reshape(-1, crop_h, crop_w, 3)[:frame_count]


def parse_args():
    parser = argparse.ArgumentParser(description=__doc__)
    source = parser.add_mutually_exclusive_group(required=True)
    source.add_argument("--jsonl_path", "--json_path", dest="json_path")
    source.add_argument("--pkl_path")
    parser.add_argument("--root_path", "--s3_data_root", default="")
    parser.add_argument("--outdir", "--output_dir", "--save_path", dest="save_path", required=True)
    parser.add_argument("--output_name", default="embeddings.jsonl")
    parser.add_argument("--start_index", type=int, default=0)
    parser.add_argument("--end_index", type=int, default=-1)
    parser.add_argument("--fps", type=int, default=24)
    parser.add_argument("--max_frames_limit", "--for_get_frames_num", type=int, default=121)
    parser.add_argument("--image_size", choices=tuple(VAE_TARGET_SIZES), default="480p")
    parser.add_argument("--ffmpeg_dir", default="", help="空值使用 PATH；与 VAE 保持同一 FFmpeg 版本")
    parser.add_argument("--config_path", required=True)
    parser.add_argument("--ckpt_path", required=True)
    parser.add_argument("--project_root", default="", help="包含 mimogpt 的项目根目录")
    parser.add_argument("--backend", choices=("nccl", "hccl", "gloo"), default=None)
    parser.add_argument("--model_name", default="vae-1.2")
    parser.add_argument("--using_tiling", default="0")
    parser.add_argument("--height", type=int, default=720)
    parser.add_argument("--width", type=int, default=480)
    parser.add_argument("--max_frame", type=int, default=121)
    parser.add_argument("--task", default="")
    parser.add_argument("--dataset_name", default="one_video")
    parser.add_argument("--global-seed", type=int, default=0)
    parser.add_argument("--num_workers", type=int, default=0)
    parser.add_argument("--max_time", type=int, default=16)
    parser.add_argument("--video_spec", default="")
    parser.add_argument("--pre_type", default="tensor")
    args = parser.parse_args()
    if args.fps <= 0 or args.max_frames_limit < 73:
        parser.error("fps 必须大于 0，max_frames_limit 必须至少为 73")
    if args.start_index < 0 or args.end_index < -1:
        parser.error("无效的数据切片范围")
    if args.num_workers != 0:
        parser.error("分布式协同编码当前要求 --num_workers 0")
    if "://" in args.save_path:
        parser.error("outdir 必须是本地目录")
    if Path(args.output_name).name != args.output_name:
        parser.error("output_name 必须是文件名")
    # 与 vae.py 的 encoder 参数结构保持兼容。所有 rank 使用同一数据集。
    args.gpus, args.global_rank = 1, 0
    args.is_t2v = 1
    args.is_i2v = args.is_i2v_mid = args.is_i2v_end = args.is_v2v = 0
    args.bucket_path = ""
    return args


@contextmanager
def local_file(path):
    if path.startswith(("s3://", "obs://")):
        import moxing as mox
        with tempfile.TemporaryDirectory(prefix="vae_input_") as cache:
            local = os.path.join(cache, path.rsplit("/", 1)[-1])
            mox.file.copy(path, local)
            yield local
    else:
        yield path


def feature_relative_path(video_fn):
    """保留视频目录和扩展名，避免 a.mp4 与 a.avi 的特征重名。"""
    path = PurePosixPath(video_fn)
    if not video_fn or path.is_absolute() or ".." in path.parts or ":" in video_fn or "\\" in video_fn or not path.name:
        raise ValueError(f"video_fn 必须为安全的相对路径: {video_fn!r}")
    return str(PurePosixPath("video_latent") / (str(path) + ".pt"))


def read_records(args):
    with local_file(args.json_path or args.pkl_path) as path:
        if args.json_path:
            with open(path, encoding="utf-8") as stream:
                rows = [json.loads(line) for line in stream if line.strip()]
        else:
            with open(path, "rb") as stream:
                rows = pickle.load(stream)
    rows = rows[args.start_index:None if args.end_index == -1 else args.end_index]
    records, seen = [], set()
    for row in rows:
        row = {"video_fn": row} if isinstance(row, str) else dict(row)
        relative = feature_relative_path(row["video_fn"])
        if relative in seen:
            raise ValueError(f"重复视频输出路径: {relative}")
        seen.add(relative)
        records.append(row)
    return records


class VideoDataset:
    def __init__(self, records, args):
        self.records, self.args = records, args

    def __len__(self):
        return len(self.records)

    def __getitem__(self, index):
        import torch
        row = self.records[index]
        args = self.args
        path = os.path.join(args.root_path, row["video_fn"])
        save_path = str(Path(args.save_path) / feature_relative_path(row["video_fn"]))
        try:
            with local_file(path) as local:
                frames = load_video(local, args.max_frames_limit, args.fps, args.image_size, args.ffmpeg_dir)
            # [T,H,W,C] uint8 -> [C,T,H,W] float32 in [-1,1].
            tensor = torch.from_numpy(frames.copy()).permute(3, 0, 1, 2).float()
            tensor = tensor / 127.5 - 1.0
            return tensor, save_path, json.dumps(row, ensure_ascii=False)
        except Exception as exc:
            print(f"Failed to load {row['video_fn']}: {exc}", flush=True)
            return "None"


def build_encoder(args, dataloader):
    # 保持 vae.py 的构造接口，后续在此替换 MiniMax encoder。
    if args.project_root:
        sys.path.insert(0, os.path.abspath(args.project_root))
    from mimogpt.models.modules.ldm.encoder.distributed_vae import DistributedVAE
    return DistributedVAE(args, dataloader=dataloader)


def save_embedding(embedding, row, outdir, manifest):
    import torch
    relative = feature_relative_path(row["video_fn"])
    destination = Path(outdir) / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    # 不覆盖已有特征；写完特征后才追加索引。
    with destination.open("xb") as stream:
        torch.save(embedding, stream)
    result = dict(row)
    result["vae_fn"] = relative
    result["4_vae_feature_shape"] = list(embedding.shape)
    result["4_vae_feature_length"] = embedding.shape[0]
    manifest.write(json.dumps(result, ensure_ascii=False) + "\n")
    manifest.flush()
    return result


def run_inference(args):
    import torch
    import torch.distributed as dist
    from torch.utils.data import DataLoader
    npu_available = False
    try:
        import torch_npu
        npu_available = torch.npu.is_available()
        if npu_available:
            from torch_npu.contrib import transfer_to_npu
    except ImportError:
        pass
    args.rank = int(os.environ.get("RANK", "0"))
    args.world_size = int(os.environ.get("WORLD_SIZE", "1"))
    args.local_rank = int(os.environ.get("LOCAL_RANK", "0"))
    args.backend = args.backend or ("hccl" if npu_available else "nccl")
    device = torch.device(f"{'npu' if npu_available else 'cuda'}:{args.local_rank}")
    if npu_available:
        torch.npu.set_device(device)
    else:
        torch.cuda.set_device(device)
    torch.manual_seed(args.global_seed)
    records = read_records(args)
    dataloader = DataLoader(VideoDataset(records, args), batch_size=1,
                            shuffle=False, num_workers=0, drop_last=False)
    outdir = Path(args.save_path)
    manifest = None
    with tempfile.TemporaryDirectory(prefix="vae_dist_") as rendezvous:
        init_method = "env://" if "RANK" in os.environ else Path(rendezvous, "init").as_uri()
        dist.init_process_group(args.backend, init_method=init_method,
                                rank=args.rank, world_size=args.world_size,
                                timeout=datetime.timedelta(hours=2))
        try:
            error = None
            if args.rank == 0:
                try:
                    outdir.mkdir(parents=True, exist_ok=True)
                    for row in records:
                        if (outdir / feature_relative_path(row["video_fn"])).exists():
                            raise FileExistsError(feature_relative_path(row["video_fn"]))
                    manifest = (outdir / args.output_name).open("x", encoding="utf-8")
                except Exception as exc:
                    error = str(exc)
            status = [error]
            dist.broadcast_object_list(status, src=0)
            if status[0] is not None:
                raise RuntimeError(f"Output initialization failed: {status[0]}")
            encoder = build_encoder(args, dataloader)
            saved, skipped = 0, 0
            for batch in dataloader:
                failed = isinstance(batch[0], str) and batch[0] == "None"
                flag = torch.tensor([int(failed)], device=device)
                dist.all_reduce(flag)
                if flag.item():
                    skipped += 1
                    continue
                row = json.loads(batch[2][0])
                x = batch[0].to(device=device, dtype=torch.bfloat16)
                with torch.no_grad():
                    moments = encoder.encode(x)[0]
                    embedding = moments.detach().to(device="cpu", dtype=torch.bfloat16)
                    embedding = embedding.permute(0, 2, 1, 3, 4)[0].contiguous()
                status = [None]
                if args.rank == 0:
                    try:
                        save_embedding(embedding, row, outdir, manifest)
                    except Exception as exc:
                        status[0] = str(exc)
                dist.broadcast_object_list(status, src=0)
                if status[0] is not None:
                    raise RuntimeError(f"Saving {row['video_fn']} failed: {status[0]}")
                saved += 1
                if args.rank == 0:
                    print(f"Saved {saved}/{len(records)}: {row['video_fn']}", flush=True)
            if args.rank == 0:
                print(f"Finished: saved={saved}, skipped={skipped}, index={outdir / args.output_name}")
        finally:
            if manifest is not None:
                manifest.close()
            dist.destroy_process_group()


if __name__ == "__main__":
    run_inference(parse_args())
