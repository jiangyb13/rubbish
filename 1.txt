def load_and_prepare(filepath):
    dataframe = pd.read_json(filepath, orient="records", lines=True)
    dataframe["meta_key"] = dataframe["metadata"].apply(
        lambda value: (
            json.dumps(value, sort_keys=True) if isinstance(value, dict) else value
        )
    )
    dataframe["sample_id"] = dataframe["filename"].apply(os.path.basename)
    if "tag" not in dataframe:
        dataframe["tag"] = dataframe["metadata"].apply(
            lambda value: value.get("tag") if isinstance(value, dict) else None
        )
    required_columns = {"filename", "metadata", "tag", "correct"}
    missing_columns = required_columns.difference(dataframe.columns)
    if missing_columns:
        raise ValueError(
            f"{filepath} is missing required columns: {sorted(missing_columns)}"
        )
    return dataframe


def make_index(dataframe, name):
    keys = list(zip(dataframe["meta_key"], dataframe["sample_id"]))
    duplicate_count = len(keys) - len(set(keys))
    if duplicate_count:
        raise ValueError(f"{name} contains {duplicate_count} duplicate sample keys")
    return {key: row for key, (_, row) in zip(keys, dataframe.iterrows())}


def summarize_geneval(dataframe):
    task_scores = {
        str(tag): float(task_df["correct"].astype(bool).mean())
        for tag, task_df in dataframe.groupby("tag", sort=False)
    }
    return {
        "num_images": int(len(dataframe)),
        "num_prompts": int(dataframe["meta_key"].nunique()),
        "correct_images": int(dataframe["correct"].astype(bool).sum()),
        "image_accuracy": float(dataframe["correct"].astype(bool).mean()),
        "task_scores": task_scores,
        "overall": float(np.mean(list(task_scores.values()))),
    }


def print_summary(name, summary):
    task_text = "  ".join(
        f"{tag}={score:.2%}" for tag, score in summary["task_scores"].items()
    )
    print(
        f"{name}: overall={summary['overall']:.5f}, "
        f"image_acc={summary['image_accuracy']:.2%} "
        f"({summary['correct_images']}/{summary['num_images']})"
    )
    print(f"  {task_text}")


def json_default(value):
    if isinstance(value, np.generic):
        return value.item()
    raise TypeError(f"Object of type {type(value).__name__} is not JSON serializable")


class BagelCEScorer:
    def __init__(self, model_path, ce_max_tokens, vit_max_side, score_mode):
        (
            self.model,
            self.vae_model,
            self.tokenizer,
            self.new_token_ids,
            _,
            self.vit_transform,
        ) = initialize_full_model(model_path)
        self.vae_model = self.vae_model.to("cpu")
        self.ce_max_tokens = ce_max_tokens
        self.vit_max_side = vit_max_side
        self.score_mode = score_mode
        self.cache = {}
        vit_parameter = next(self.model.vit_model.parameters())
        self.vit_device = vit_parameter.device
        self.vit_dtype = vit_parameter.dtype
        for parameter in self.model.parameters():
            parameter.requires_grad_(False)

    @staticmethod
    def image_to_tensor(image):
        array = np.asarray(image.convert("RGB"), dtype=np.float32) / 255.0
        return (
            torch.from_numpy(array)
            .permute(2, 0, 1)
            .unsqueeze(0)
            .contiguous()
        )

    def score(self, image_path, prompt):
        cache_key = (image_path, prompt, self.score_mode)
        if cache_key in self.cache:
            return self.cache[cache_key]
        if not os.path.exists(image_path):
            raise FileNotFoundError(f"Image does not exist: {image_path}")

        try:
            with Image.open(image_path) as image:
                width, height = image.size
                image_tensor = self.image_to_tensor(image)
            image_tensor = image_tensor.to(
                device=self.vit_device,
                dtype=self.vit_dtype,
            )
            common_kwargs = {
                "x_t_0": None,
                "vae_model": self.vae_model,
                "image_shape": (height, width),
                "image_transform": self.vit_transform,
                "tokenizer": self.tokenizer,
                "new_token_ids": self.new_token_ids,
                "ce_vit_max_side": self.vit_max_side,
                "umm_dropbp_layers": (),
                "decoded_image": image_tensor,
            }
            with torch.no_grad():
                caption_score = None
                binary_score = None
                if self.score_mode in ("caption", "dual_consensus"):
                    loss, _ = self.model.UnderstandingCELoss(
                        target_text=prompt,
                        ce_max_tokens=self.ce_max_tokens,
                        **common_kwargs,
                    )
                    caption_score = loss.float().item()
                if self.score_mode in ("binary_logodds", "dual_consensus"):
                    instruction = (
                        "Act as a strict visual verifier. Inspect the image and "
                        "decide whether it satisfies every explicit requirement "
                        "in the target prompt. Check object identities, exact "
                        "counts, colors, attribute bindings, and spatial "
                        "relations. If any required detail is missing, wrong, "
                        "ambiguous, or not visibly supported, answer No. "
                        f"Target prompt: {prompt}\n"
                        "Answer exactly Yes or No."
                    )
                    yes_loss, _ = self.model.UnderstandingCELoss(
                        target_text="Yes",
                        instruction=instruction,
                        ce_max_tokens=1,
                        **common_kwargs,
                    )
                    no_loss, _ = self.model.UnderstandingCELoss(
                        target_text="No",
                        instruction=instruction,
                        ce_max_tokens=1,
                        **common_kwargs,
                    )
                    # Lower is better: negative values favor Yes over No.
                    binary_score = (yes_loss.float() - no_loss.float()).item()
                if self.score_mode == "caption":
                    score = caption_score
                elif self.score_mode == "binary_logodds":
                    score = binary_score
                else:
                    score = {
                        "caption": caption_score,
                        "binary": binary_score,
                    }
            self.cache[cache_key] = score
            return score
        except Exception as error:
            raise RuntimeError(f"Failed to score {image_path}: {error}") from error


class LongCLIPScorer:
    def __init__(self, model_path):
        from modeling.longclip_b.model import longclip

        if not os.path.exists(model_path):
            raise FileNotFoundError(f"LongCLIP checkpoint does not exist: {model_path}")
        self.longclip = longclip
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.model, self.preprocess = longclip.load(model_path, device=self.device)
        self.model.eval()
        self.cache = {}

    def score(self, image_path, prompt):
        cache_key = (image_path, prompt)
        if cache_key in self.cache:
            return self.cache[cache_key]
        if not os.path.exists(image_path):
            raise FileNotFoundError(f"Image does not exist: {image_path}")

        try:
            with Image.open(image_path) as image:
                image_tensor = self.preprocess(image.convert("RGB")).unsqueeze(0)
            image_tensor = image_tensor.to(self.device)
            text_tensor = self.longclip.tokenize([prompt]).to(self.device)
            with torch.inference_mode():
                image_features = self.model.encode_image(image_tensor)
                text_features = self.model.encode_text(text_tensor)
                image_features = image_features / image_features.norm(
                    dim=-1, keepdim=True
                ).clamp_min(1e-12)
                text_features = text_features / text_features.norm(
                    dim=-1, keepdim=True
                ).clamp_min(1e-12)
                similarity = (image_features @ text_features.T).float().item()
            self.cache[cache_key] = similarity
            return similarity
        except Exception as error:
            raise RuntimeError(f"Failed to score {image_path}: {error}") from error


def select_results(ordered_keys, index_a, index_b, scorer, prefer_lower, name):
    final_results = []
    stats = {
        "agreements_skipped": 0,
        "decisions": 0,
        "a_selected": 0,
        "b_selected": 0,
        "selected_correct": 0,
    }
    for key in tqdm(ordered_keys, total=len(ordered_keys), desc=f"Selecting by {name}"):
        row_a = index_a[key]
        row_b = index_b[key]
        correct_a = bool(row_a["correct"])
        correct_b = bool(row_b["correct"])

        if correct_a == correct_b:
            # This choice cannot change GenEval, so avoid scoring both images.
            selected_row = row_a.to_dict()
            stats["agreements_skipped"] += 1
            stats["a_selected"] += 1
        else:
            metadata = json.loads(key[0]) if isinstance(key[0], str) else key[0]
            prompt = metadata["prompt"]
            score_a = scorer.score(row_a["filename"], prompt)
            score_b = scorer.score(row_b["filename"], prompt)
            choose_a = score_a <= score_b if prefer_lower else score_a >= score_b
            selected_row = (row_a if choose_a else row_b).to_dict()
            stats["a_selected" if choose_a else "b_selected"] += 1
            stats["decisions"] += 1
            stats["selected_correct"] += int(bool(selected_row["correct"]))

        selected_row.pop("meta_key", None)
        selected_row.pop("sample_id", None)
        final_results.append(selected_row)
    return final_results, stats


def new_selection_stats():
    return {
        "agreements_skipped": 0,
        "decisions": 0,
        "a_selected": 0,
        "b_selected": 0,
        "selected_correct": 0,
    }


def select_results_dual_ce(
    ordered_keys,
    index_a,
    index_b,
    scorer,
    binary_weight,
    caption_margin_scale=None,
    binary_margin_scale=None,
):
    if not 0.0 <= binary_weight <= 1.0:
        raise ValueError("binary_weight must be between 0 and 1")

    rows = {
        "caption": [],
        "binary": [],
        "dual": [],
        "fused": [],
        "confidence": [],
    }
    stats = {name: new_selection_stats() for name in rows}
    stats["dual"].update(
        {
            "verifier_agreements": 0,
            "verifier_disagreements": 0,
        }
    )

    score_cache = {}
    if (caption_margin_scale is None) != (binary_margin_scale is None):
        raise ValueError(
            "caption_margin_scale and binary_margin_scale must be provided together"
        )
    decision_keys = [
        key
        for key in ordered_keys
        if bool(index_a[key]["correct"]) != bool(index_b[key]["correct"])
    ]
    for key in tqdm(
        decision_keys,
        total=len(decision_keys),
        desc="Scoring dual BAGEL CE",
    ):
        row_a = index_a[key]
        row_b = index_b[key]
        metadata = json.loads(key[0]) if isinstance(key[0], str) else key[0]
        prompt = metadata["prompt"]
        scores_a = scorer.score(row_a["filename"], prompt)
        scores_b = scorer.score(row_b["filename"], prompt)
        caption_margin = scores_b["caption"] - scores_a["caption"]
        binary_margin = scores_b["binary"] - scores_a["binary"]
        score_cache[key] = {
            "caption_a": scores_a["caption"],
            "caption_b": scores_b["caption"],
            "binary_a": scores_a["binary"],
            "binary_b": scores_b["binary"],
            "caption_margin": caption_margin,
            "binary_margin": binary_margin,
        }

    eps = 1e-12
    use_fixed_scales = caption_margin_scale is not None
    if use_fixed_scales:
        caption_margin_scale = max(float(caption_margin_scale), eps)
        binary_margin_scale = max(float(binary_margin_scale), eps)
        print(
            "Dual-CE calibration: fixed validation scales, "
            f"caption_scale={caption_margin_scale:.6g}, "
            f"binary_scale={binary_margin_scale:.6g}, "
            f"binary_weight={binary_weight:.3f}"
        )
    else:
        print(
            "Dual-CE calibration: label-free pair-relative margins, "
            f"binary_weight={binary_weight:.3f}"
        )

    for key in tqdm(
        ordered_keys,
        total=len(ordered_keys),
        desc="Fusing dual BAGEL CE",
    ):
        row_a = index_a[key]
        row_b = index_b[key]
        correct_a = bool(row_a["correct"])
        correct_b = bool(row_b["correct"])

        if correct_a == correct_b:
            selected = {name: row_a for name in rows}
            for name in rows:
                stats[name]["agreements_skipped"] += 1
                stats[name]["a_selected"] += 1
        else:
            caption_margin = score_cache[key]["caption_margin"]
            binary_margin = score_cache[key]["binary_margin"]
            if use_fixed_scales:
                normalized_caption = caption_margin / caption_margin_scale
                normalized_binary = binary_margin / binary_margin_scale
            else:
                caption_pair_scale = max(
                    0.5
                    * (
                        abs(score_cache[key]["caption_a"])
                        + abs(score_cache[key]["caption_b"])
                    ),
                    eps,
                )
                binary_pair_scale = max(
                    0.5
                    * (
                        abs(score_cache[key]["binary_a"])
                        + abs(score_cache[key]["binary_b"])
                    ),
                    eps,
                )
                normalized_caption = caption_margin / caption_pair_scale
                normalized_binary = binary_margin / binary_pair_scale
            caption_choose_a = caption_margin >= 0.0
            binary_choose_a = binary_margin >= 0.0
            dual_choose_a = (
                caption_choose_a
                if caption_choose_a == binary_choose_a
                else True
            )
            fused_margin = (
                (1.0 - binary_weight) * normalized_caption
                + binary_weight * normalized_binary
            )
            fused_choose_a = fused_margin >= 0.0
            if caption_choose_a == binary_choose_a:
                confidence_choose_a = caption_choose_a
            elif abs(normalized_caption) > abs(normalized_binary):
                confidence_choose_a = caption_choose_a
            elif abs(normalized_binary) > abs(normalized_caption):
                confidence_choose_a = binary_choose_a
            else:
                confidence_choose_a = True
            choices = {
                "caption": caption_choose_a,
                "binary": binary_choose_a,
                "dual": dual_choose_a,
                "fused": fused_choose_a,
                "confidence": confidence_choose_a,
            }
            selected = {
                name: row_a if choose_a else row_b
                for name, choose_a in choices.items()
            }
            for name, choose_a in choices.items():
                stats[name]["decisions"] += 1
                stats[name]["a_selected" if choose_a else "b_selected"] += 1
                stats[name]["selected_correct"] += int(
                    bool(selected[name]["correct"])
                )
            if caption_choose_a == binary_choose_a:
                stats["dual"]["verifier_agreements"] += 1
            else:
                stats["dual"]["verifier_disagreements"] += 1

        for name, selected_row in selected.items():
            clean_row = selected_row.to_dict()
            clean_row.pop("meta_key", None)
            clean_row.pop("sample_id", None)
            rows[name].append(clean_row)

    stats["fused"].update(
        {
            "caption_margin_scale": caption_margin_scale,
            "binary_margin_scale": binary_margin_scale,
            "binary_weight": binary_weight,
            "normalization": "fixed_validation" if use_fixed_scales else "pair_relative",
        }
    )
    stats["confidence"].update(
        {
            "caption_margin_scale": caption_margin_scale,
            "binary_margin_scale": binary_margin_scale,
            "normalization": "fixed_validation" if use_fixed_scales else "pair_relative",
        }
    )

    return rows, stats


def save_results(entries, path):
    with open(path, "w", encoding="utf-8") as output_file:
        for entry in entries:
            output_file.write(
                json.dumps(entry, ensure_ascii=False, default=json_default) + "\n"
            )


def default_clip_output_path(ce_output_path):
    stem, extension = os.path.splitext(ce_output_path)
    return f"{stem}_clip{extension or '.jsonl'}"


def default_variant_output_path(output_path, variant):
    stem, extension = os.path.splitext(output_path)
    return f"{stem}_{variant}{extension or '.jsonl'}"


def release_scorer(scorer):
    del scorer
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    return None


def main():
    args = parse_args()
    print("Reading result files A and B...")
    df_a = load_and_prepare(args.file_a)
    df_b = load_and_prepare(args.file_b)
    index_a = make_index(df_a, "A")
    index_b = make_index(df_b, "B")
    keys_a = set(index_a)
    keys_b = set(index_b)
    if keys_a != keys_b:
        raise ValueError(
            "A and B must contain the same GenEval samples for a fair comparison: "
            f"A-only={len(keys_a - keys_b)}, B-only={len(keys_b - keys_a)}"
        )

    summary_a = summarize_geneval(df_a)
    summary_b = summarize_geneval(df_b)
    print("\nOriginal GenEval results")
    print_summary("A", summary_a)
    print_summary("B", summary_b)

    disagreement_count = sum(
        bool(index_a[key]["correct"]) != bool(index_b[key]["correct"])
        for key in keys_a
    )
    print(
        f"\nCorrectness agreement: {len(keys_a) - disagreement_count}/{len(keys_a)}; "
        f"CE decisions needed: {disagreement_count}"
    )

    ordered_keys = list(zip(df_a["meta_key"], df_a["sample_id"]))
    selector_results = {}

    if args.selector in ("clip", "both"):
        clip_output_path = args.clip_output_file or default_clip_output_path(
            args.output_file
        )
        if disagreement_count:
            print("Loading LongCLIP model...")
            clip_scorer = LongCLIPScorer(args.longclip_path)
            clip_rows, clip_stats = select_results(
                ordered_keys,
                index_a,
                index_b,
                clip_scorer,
                prefer_lower=False,
                name="LongCLIP",
            )
            clip_scorer = release_scorer(clip_scorer)
        else:
            clip_rows, clip_stats = select_results(
                ordered_keys, index_a, index_b, None, False, "LongCLIP"
            )
        save_results(clip_rows, clip_output_path)
        clip_summary = summarize_geneval(load_and_prepare(clip_output_path))
        selector_results["clip_selected"] = {
            "summary": clip_summary,
            "gain_vs_a": clip_summary["overall"] - summary_a["overall"],
            "gain_vs_b": clip_summary["overall"] - summary_b["overall"],
            "selection": clip_stats,
            "output": clip_output_path,
        }

    if args.selector in ("ce", "both"):
        if disagreement_count:
            print("Loading BAGEL understanding model...")
            ce_scorer = BagelCEScorer(
                args.model_path,
                args.ce_max_tokens,
                args.vit_max_side,
                args.ce_score_mode,
            )
            if args.ce_score_mode == "dual_consensus":
                dual_rows, dual_stats = select_results_dual_ce(
                    ordered_keys,
                    index_a,
                    index_b,
                    ce_scorer,
                    args.binary_weight,
                    args.caption_margin_scale,
                    args.binary_margin_scale,
                )
            else:
                ce_rows, ce_stats = select_results(
                    ordered_keys,
                    index_a,
                    index_b,
                    ce_scorer,
                    prefer_lower=True,
                    name="BAGEL CE",
                )
            ce_scorer = release_scorer(ce_scorer)
        else:
            if args.ce_score_mode == "dual_consensus":
                dual_rows, dual_stats = select_results_dual_ce(
                    ordered_keys,
                    index_a,
                    index_b,
                    None,
                    args.binary_weight,
                    args.caption_margin_scale,
                    args.binary_margin_scale,
                )
            else:
                ce_rows, ce_stats = select_results(
                    ordered_keys, index_a, index_b, None, True, "BAGEL CE"
                )

        if args.ce_score_mode == "dual_consensus":
            caption_output_path = (
                args.caption_output_file
                or default_variant_output_path(args.output_file, "caption")
            )
            binary_output_path = (
                args.binary_output_file
                or default_variant_output_path(args.output_file, "binary")
            )
            fusion_output_path = (
                args.fusion_output_file
                or default_variant_output_path(args.output_file, "fused")
            )
            confidence_output_path = (
                args.confidence_output_file
                or default_variant_output_path(args.output_file, "confidence")
            )
            output_paths = {
                "caption_ce_selected": caption_output_path,
                "binary_ce_selected": binary_output_path,
                "dual_ce_selected": args.output_file,
                "fused_ce_selected": fusion_output_path,
                "confidence_ce_selected": confidence_output_path,
            }
            row_names = {
                "caption_ce_selected": "caption",
                "binary_ce_selected": "binary",
                "dual_ce_selected": "dual",
                "fused_ce_selected": "fused",
                "confidence_ce_selected": "confidence",
            }
            for result_name, output_path in output_paths.items():
                row_name = row_names[result_name]
                save_results(dual_rows[row_name], output_path)
                result_summary = summarize_geneval(load_and_prepare(output_path))
                selector_results[result_name] = {
                    "summary": result_summary,
                    "gain_vs_a": result_summary["overall"] - summary_a["overall"],
                    "gain_vs_b": result_summary["overall"] - summary_b["overall"],
                    "selection": dual_stats[row_name],
                    "output": output_path,
                }
        else:
            save_results(ce_rows, args.output_file)
            ce_summary = summarize_geneval(load_and_prepare(args.output_file))
            selector_results["ce_selected"] = {
                "summary": ce_summary,
                "gain_vs_a": ce_summary["overall"] - summary_a["overall"],
                "gain_vs_b": ce_summary["overall"] - summary_b["overall"],
                "selection": ce_stats,
                "output": args.output_file,
            }

    metrics = {
        "a": summary_a,
        "b": summary_b,
        "ce_score_mode": args.ce_score_mode,
        **selector_results,
    }
    metrics_path = args.metrics_file or f"{args.output_file}.metrics.json"
    with open(metrics_path, "w", encoding="utf-8") as metrics_file:
        json.dump(metrics, metrics_file, indent=2, ensure_ascii=False)

    print("\n" + "=" * 50)
    print("Final GenEval comparison")
    print_summary("A", summary_a)
    print_summary("B", summary_b)
    display_names = {
        "clip_selected": "LongCLIP selected",
        "ce_selected": "CE selected",
        "caption_ce_selected": "Caption CE selected",
        "binary_ce_selected": "Binary CE selected",
        "dual_ce_selected": "Dual-CE consensus",
        "fused_ce_selected": "Normalized CE fusion",
        "confidence_ce_selected": "Confidence-routed CE",
    }
    for result_name, result in selector_results.items():
        display_name = display_names[result_name]
        print_summary(display_name, result["summary"])
        print(
            f"  Gain vs A: {result['gain_vs_a']:+.5f} "
            f"({result['gain_vs_a'] * 100:+.2f} points)"
        )
        print(
            f"  Gain vs B: {result['gain_vs_b']:+.5f} "
            f"({result['gain_vs_b'] * 100:+.2f} points)"
        )
        selection = result["selection"]
        if selection["decisions"]:
            decision_accuracy = selection["selected_correct"] / selection["decisions"]
            print(
                f"  Disagreement accuracy: {decision_accuracy:.2%} "
                f"({selection['selected_correct']}/{selection['decisions']})"
            )
        print(f"  Output: {result['output']}")
    print(f"Metrics: {metrics_path}")
    print("=" * 50)


if __name__ == "__main__":
    main()
