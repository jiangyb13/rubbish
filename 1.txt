#!/usr/bin/env python3
"""Merge a rectified candidate with Base using two native BAGEL CE views.

The first JSONL is the rectified candidate and the second is Base.  Lower CE
is better.  The candidate is kept unless both the user-caption CE and a strict
Correct/Incorrect CE verifier prefer Base.  The rule is global and never uses
the GenEval task/tag.

For evaluation efficiency, pairs with identical ``correct`` labels are skipped
by default because their choice cannot change the reported score.  Pass
``--score-all`` to score every pair for deployment-style selection.
"""

from __future__ import annotations

import argparse
import inspect
import json
import os
from collections import defaultdict
from pathlib import Path

import numpy as np
from PIL import Image
import torch
from tqdm import tqdm


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("candidate_jsonl", type=Path)
    parser.add_argument("base_jsonl", type=Path)
    parser.add_argument("-o", "--output", type=Path, required=True)
    parser.add_argument("--metrics-file", type=Path)
    parser.add_argument(
        "--model-path",
        default="/data/jyb/pretrained_models/BAGEL-7B-MoT",
    )
    parser.add_argument("--caption-vit-max-side", type=int, default=168)
    parser.add_argument("--correct-vit-max-side", type=int, default=224)
    parser.add_argument("--caption-max-tokens", type=int, default=192)
    parser.add_argument(
        "--score-all",
        action="store_true",
        help="Score all pairs instead of skipping equal-correctness pairs.",
    )
    return parser.parse_args()


def read_jsonl(path: Path) -> list[dict]:
    with path.open("r", encoding="utf-8") as handle:
        return [json.loads(line) for line in handle if line.strip()]


def metadata_dict(value):
    parsed = json.loads(value) if isinstance(value, str) else value
    if not isinstance(parsed, dict):
        raise ValueError("metadata must be a dict or a JSON object string")
    return parsed


def row_key(row: dict) -> tuple[str, str]:
    return (
        json.dumps(metadata_dict(row["metadata"]), sort_keys=True, ensure_ascii=False),
        os.path.basename(row["filename"]),
    )


def summarize(rows: list[dict]) -> dict:
    by_tag = defaultdict(list)
    for row in rows:
        by_tag[str(row["tag"])].append(bool(row["correct"]))
    per_tag = {
        tag: sum(values) / len(values) for tag, values in sorted(by_tag.items())
    }
    correct = sum(bool(row["correct"]) for row in rows)
    return {
        "macro": sum(per_tag.values()) / len(per_tag),
        "micro": correct / len(rows),
        "correct": correct,
        "total": len(rows),
        "per_tag": per_tag,
    }


class DualViewCEScorer:
    def __init__(self, args: argparse.Namespace):
        import inference
        from modeling.bagel.b import Bagel as CEBagel
        from modeling.bagel.b import BagelConfig as CEBagelConfig

        inference.Bagel = CEBagel
        inference.BagelConfig = CEBagelConfig
        (
            self.model,
            self.vae,
            self.tokenizer,
            self.token_ids,
            _,
            self.image_transform,
        ) = inference.initialize_full_model(args.model_path)
        if not hasattr(self.model, "UnderstandingCELoss"):
            raise RuntimeError("Loaded BAGEL model has no UnderstandingCELoss")

        self.vae = self.vae.to("cpu")
        self.device = next(self.model.vit_model.parameters()).device
        self.dtype = next(self.model.vit_model.parameters()).dtype
        self.caption_side = args.caption_vit_max_side
        self.correct_side = args.correct_vit_max_side
        self.caption_tokens = args.caption_max_tokens
        self.supports_context = (
            "prepared_visual_context"
            in inspect.signature(self.model.UnderstandingCELoss).parameters
        )
        if not self.supports_context:
            raise RuntimeError(
                "This script requires UnderstandingCELoss visual-context reuse"
            )
        for parameter in self.model.parameters():
            parameter.requires_grad_(False)

    def image_tensor(self, path: str) -> tuple[torch.Tensor, tuple[int, int]]:
        if not os.path.isfile(path):
            raise FileNotFoundError(path)
        with Image.open(path) as image:
            image = image.convert("RGB")
            width, height = image.size
            array = np.asarray(image, dtype=np.float32) / 255.0
        tensor = torch.from_numpy(array).permute(2, 0, 1).unsqueeze(0).contiguous()
        return tensor.to(device=self.device, dtype=self.dtype), (height, width)

    def score(self, image_path: str, prompt: str) -> dict[str, float]:
        image, image_shape = self.image_tensor(image_path)
        common = {
            "x_t_0": None,
            "vae_model": self.vae,
            "image_shape": image_shape,
            "image_transform": self.image_transform,
            "tokenizer": self.tokenizer,
            "new_token_ids": self.token_ids,
            "umm_dropbp_layers": (),
            "decoded_image": image,
        }
        correct_instruction = (
            "Act as a strict visual verifier. Inspect the image and decide whether it "
            "satisfies every explicit requirement in the target prompt. Check object "
            "identities, exact counts, colors, attribute bindings, and spatial "
            "relations. If any required detail is missing, wrong, ambiguous, or not "
            f"visibly supported, answer Incorrect. Target prompt: {prompt}\n"
            "Answer exactly Correct or Incorrect."
        )

        with torch.inference_mode():
            caption_loss, _ = self.model.UnderstandingCELoss(
                target_text=prompt,
                instruction=None,
                ce_max_tokens=self.caption_tokens,
                ce_vit_max_side=self.caption_side,
                **common,
            )
            correct_positive, _, visual_context = self.model.UnderstandingCELoss(
                target_text="Correct",
                instruction=correct_instruction,
                ce_max_tokens=1,
                ce_vit_max_side=self.correct_side,
                return_visual_context=True,
                **common,
            )
            correct_negative, _ = self.model.UnderstandingCELoss(
                target_text="Incorrect",
                instruction=correct_instruction,
                ce_max_tokens=1,
                ce_vit_max_side=self.correct_side,
                prepared_visual_context=visual_context,
                **common,
            )
        return {
            "caption": float(caption_loss.float().item()),
            "correct": float(
                (correct_positive.float() - correct_negative.float()).item()
            ),
        }


def main() -> None:
    args = parse_args()
    candidate_rows = read_jsonl(args.candidate_jsonl)
    base_rows = read_jsonl(args.base_jsonl)
    candidate = {row_key(row): row for row in candidate_rows}
    base = {row_key(row): row for row in base_rows}
    if len(candidate) != len(candidate_rows) or len(base) != len(base_rows):
        raise RuntimeError("A JSONL contains duplicate metadata/sample keys")
    if set(candidate) != set(base):
        raise RuntimeError("Candidate and Base JSONLs contain different samples")

    ordered_keys = [row_key(row) for row in base_rows]
    score_keys = [
        key
        for key in ordered_keys
        if args.score_all
        or bool(candidate[key]["correct"]) != bool(base[key]["correct"])
    ]
    scorer = DualViewCEScorer(args) if score_keys else None

    scores = {}
    for key in tqdm(score_keys, desc="Scoring candidate/Base with native UMM CE"):
        prompt = candidate[key].get("prompt") or metadata_dict(
            candidate[key]["metadata"]
        ).get("prompt")
        if not isinstance(prompt, str) or not prompt.strip():
            raise ValueError(f"Missing prompt for sample key: {key}")
        candidate_score = scorer.score(candidate[key]["filename"], prompt)
        base_score = scorer.score(base[key]["filename"], prompt)
        scores[key] = {
            "candidate": candidate_score,
            "base": base_score,
            "delta_caption": candidate_score["caption"] - base_score["caption"],
            "delta_correct": candidate_score["correct"] - base_score["correct"],
        }

    merged = []
    base_selected = fixes = harms = 0
    for key in ordered_keys:
        candidate_row = candidate[key]
        base_row = base[key]
        if key not in scores:
            use_base = False
            selector = {
                "selected": "candidate",
                "reason": "equal_correctness_skip_ce",
            }
        else:
            pair_scores = scores[key]
            use_base = (
                pair_scores["delta_caption"] > 0.0
                and pair_scores["delta_correct"] > 0.0
            )
            selector = {
                "selected": "base" if use_base else "candidate",
                "reason": (
                    "both_ce_views_prefer_base"
                    if use_base
                    else "keep_candidate_without_two_view_base_consensus"
                ),
                **pair_scores,
            }
        selected = dict(base_row if use_base else candidate_row)
        selected["base_ce_selector"] = selector
        merged.append(selected)
        base_selected += int(use_base)
        fixes += int(
            (not bool(candidate_row["correct"])) and bool(selected["correct"])
        )
        harms += int(
            bool(candidate_row["correct"]) and (not bool(selected["correct"]))
        )

    args.output.parent.mkdir(parents=True, exist_ok=True)
    with args.output.open("w", encoding="utf-8") as handle:
        for row in merged:
            handle.write(json.dumps(row, ensure_ascii=False) + "\n")

    report = {
        "rule": (
            "keep candidate unless both User Caption CE and strict Correct CE "
            "prefer Base"
        ),
        "score_all": args.score_all,
        "pairs_scored": len(score_keys),
        "pairs_skipped": len(ordered_keys) - len(score_keys),
        "base_selected": base_selected,
        "fixes_vs_candidate": fixes,
        "harms_vs_candidate": harms,
        "candidate": summarize(candidate_rows),
        "base": summarize(base_rows),
        "merged": summarize(merged),
    }
    metrics_path = args.metrics_file or Path(f"{args.output}.metrics.json")
    metrics_path.parent.mkdir(parents=True, exist_ok=True)
    with metrics_path.open("w", encoding="utf-8") as handle:
        json.dump(report, handle, ensure_ascii=False, indent=2)
        handle.write("\n")
    print(json.dumps(report, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
